#!/bin/bash

# Pin the XHCI IRQs to the big cores
# Except for core 4, which we've set up to run the UVC async memcpy work
irqs=$(grep xhci-hcd /proc/interrupts | awk -F ':' '{print $1}')
for irq in $irqs; do
	echo e0 >/proc/irq/$irq/smp_affinity
done
root@gateway:/usr/sbin# cat ./rk3588-disable-spi-bootloader
#!/bin/bash

FLASH_DEV=/dev/mtd0
FLASH_BLK_DEV=/dev/mtdblock0
if [ ! -c $FLASH_DEV ]; then
	echo "$FLASH_DEV doesn't exist, exiting"
	exit
fi

if [ ! -b $FLASH_BLK_DEV ]; then
	echo "$FLASH_BLK_DEV doesn't exist, exiting"
	exit
fi

ERASE_SIZE=$(cat /sys/class/mtd/mtd0/erasesize)
if [ $ERASE_SIZE != 4096 ]; then
	echo "Unexpected erase size $ERASE_SIZE, exiting"
	exit
fi

TMP_FILE=/tmp/flash.tmp
dd if=$FLASH_BLK_DEV of=$TMP_FILE bs=$ERASE_SIZE count=1 skip=8 status=none
echo "6ae59e64850377ee5470c854761551ea $TMP_FILE" | md5sum -c &>/dev/null

if [[ $? -ne 0 ]]; then
	flash_erase $FLASH_DEV 32768 1
fi
root@gateway:/usr/sbin# cat ./rk3588-setup-leds
#!/bin/bash

set_led() {
	if [ -d "/sys/class/leds/$1" ]; then
		echo "$2" >"/sys/class/leds/$1/trigger"
	fi
}

if [ "$1" == "reset" ]; then
	set_led "blue_led" "heartbeat"
	set_led "green_led" "heartbeat"
	set_led "user-led2" "heartbeat"
else
	set_led "blue_led" "none"
	set_led "green_led" "default-on"
	set_led "user-led2" "none"
fi
